 54 def executable_underlying_price(
 55     quote: dict[str, Any] | None,
 56     *,
 57     now: datetime | None = None,
 58     max_age_seconds: int = UNDERLYING_MAX_AGE_SECONDS,
 59 ) -> tuple[float | None, str | None]:
 60     """Regular-session executable price for the live breakout trigger."""
 61     if not isinstance(quote, dict):
 62         return None, "underlying_quote_missing"
 63     bid = _positive_money(quote.get("bid_price", quote.get("bid")))
 64     ask = _positive_money(quote.get("ask_price", quote.get("ask")))

 71         or quote.get("updated_at_utc")

 78         return None, "underlying_quote_timestamp_missing"

 80         return None, "underlying_quote_stale"
 81     last = _positive_money(quote.get("last_trade_price", quote.get("last_price", quote.get("last"))))

 61 def parse_money(value: Any) -> tuple[float | None, bool]:
 62     """Return (amount, readable). readable is False when the field cannot be used."""

 64         return None, True

 70         return float(value), True
 71     if isinstance(value, str):
 72         text = value.strip().replace("$", "").replace(",", "")
 73         if text == "":
 74             return None, True
 75         try:
 76             amount = float(text)
 77         except ValueError:

 31 def dominant_bias(pattern_hits: list[dict[str, Any]]) -> str | None:
 32     for hit in pattern_hits:
 33         if hit.get("timeframe") in ("day", "daily") and hit.get("bias") in ("bullish", "bearish"):
 34             return str(hit["bias"])
 35     biases = [p.get("bias") for p in pattern_hits if p.get("bias") in ("bullish", "bearish")]
 36     if not biases:
 37         return None
 38     counts = Counter(biases)
 39     top, n = counts.most_common(1)[0]
 40     if n > len(biases) / 2:
 41         return top
 42     return None

 54 def _option_premium(quote: dict[str, Any], liq_metrics: dict[str, Any]) -> tuple[float | None, str | None]:
 55     """Playbook premium is bid/ask mid. Do not require mark_price."""
 56     mid = liq_metrics.get("mid")
 57     if mid not in (None, ""):
 58         try:
 59             value = float(mid)
 60             if value > 0:
 61                 return value, "mid"
 62         except (TypeError, ValueError):
 63             pass
 64     for key in ("mark_price", "adjusted_mark_price"):
 65         if quote.get(key) not in (None, ""):
 66             try:
 67                 value = float(quote[key])
 68                 if value > 0:
 69                     return value, key
 70             except (TypeError, ValueError):
 71                 continue
 72     return None, None

 41     h_enabled: bool | None = None,

 46     if not playbook_released:
